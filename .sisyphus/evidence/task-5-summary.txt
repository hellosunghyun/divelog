# Task 5: `/tags` Loader Optimization — COMPLETED ✓

## Objective
Optimize `getAllTags()` by merging 2 sequential queries into 1 LEFT JOIN + GROUP BY query.

## Changes Made

### File: app/db/queries/records/tags.server.ts

**Before:**
```typescript
export async function getAllTags(d1: D1Database): Promise<TagWithUsage[]> {
  const database = db(d1);

  const allTags = await database.select().from(tags).orderBy(asc(tags.name));

  const usageCounts = await database
    .select({ tagId: recordTags.tagId, count: count() })
    .from(recordTags)
    .groupBy(recordTags.tagId);

  const countMap = new Map(usageCounts.map((uc) => [uc.tagId, uc.count]));

  return allTags.map((tag) => ({
    ...tag,
    usageCount: countMap.get(tag.id) ?? 0,
  }));
}
```

**After:**
```typescript
export async function getAllTags(d1: D1Database): Promise<TagWithUsage[]> {
  const database = db(d1);

  const result = await database
    .select({
      ...getTableColumns(tags),
      usageCount: count(recordTags.tagId),
    })
    .from(tags)
    .leftJoin(recordTags, eq(tags.id, recordTags.tagId))
    .groupBy(tags.id)
    .orderBy(asc(tags.name));

  return result;
}
```

### Key Changes:
1. Added `getTableColumns` import from drizzle-orm
2. Replaced 2 sequential queries with 1 LEFT JOIN + GROUP BY
3. Eliminated JavaScript-side mapping logic
4. Preserved return type `TagWithUsage[]` (no breaking changes)
